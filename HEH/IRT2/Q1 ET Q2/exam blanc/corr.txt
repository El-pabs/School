Tâche 1 (Config commutateurs) : 

------------SWLANA1 :
en 
conf t 
vlan 10
  name compta
  exit

# si besoin : 
int vlan 10
  ip add 10.100.10.254 255.255.255.0
  exit

interface range FastEthernet0/1-2
  description To hosts VLAN 10 compta
  switchport mode access
  switchport access vlan 10
  switchport nonegotiate
  spanning-tree portfast
  spanning-tree bpduguard enable
  switchport port-security
  switchport port-security maximum 1
  switchport port-security mac-address sticky
  switchport port-security violation restrict
  no shutdown
  end
  copy running-config startup-config

---------SWWLAND 1 (ACTIF) :

  name compta
exit

  name info
exit

en

interface vlan 10
no shutdown

Tâche 4 (OSPF) : 


--------------------------------------------------RWAN 1 :

en 
conf t 
# On choisi int de sortie 
interface GigabitEthernet0/1
# priorité pour élire le DR et le BDR (routeur qui dirige et son backup)
  ip ospf priority 60
  exit
# Nbr de processus (égal à 10 dans le pdf) :
router ospf 10
# un ID pour se le rentrer dans le baltrou 
  router-id 11.11.11.11
# "Routeurs RWANx config avec OSPF Zone unique" -> Zone unique = area 0 puis on met les réseaux ciblés avec un wildcardmask 
# Réseau du routeur de base de gros con
  network 172.16.1.0 0.0.0.255 area 0 
# Réseau entre lui et RWAN 4 (aussi partagé avec Erwin 2)
  network 172.16.7.0 0.0.0.15 area 0
# Vitesse de coup de jus 
  auto-cost reference-bandwidth 10000
# "Ne doivent pas être envoyés sur les int pas utiles -> on les rend passives" 
  passiv gi 0/0
exit
