# Models for the smartcity app
from django.db import models

class TypeCapteur(models.Model):
    id_type_capteur = models.AutoField(primary_key=True)
    type = models.CharField(max_length=255)

    def __str__(self):
        return self.type


class Capteur(models.Model):
    id_capteur = models.AutoField(primary_key=True)
    nom_capteur = models.CharField(max_length=128)
    departement = models.CharField(max_length=10)
    statut = models.BooleanField()
    date_derniere_maj = models.DateField(auto_now=True)
    type_capteur = models.ForeignKey(TypeCapteur, on_delete=models.CASCADE)

    def __str__(self):
        return self.nom_capteur


class AlerteGlobale(models.Model):
    id_alerte = models.AutoField(primary_key=True)
    description = models.CharField(max_length=255)
    niveau = models.IntegerField()
    date_creation = models.DateField(auto_now_add=True)
    statut = models.BooleanField()
    id_capteur = models.ForeignKey(Capteur, on_delete=models.CASCADE)

    def __str__(self):
        return self.description


class RapportGlobal(models.Model):
    id_rapport = models.AutoField(primary_key=True)
    departement = models.CharField(max_length=10)
    periode = models.IntegerField()
    description = models.CharField(max_length=255)
    date_creation = models.DateField(auto_now_add=True)

    def __str__(self):
        return self.description


class StatistiqueGlobale(models.Model):
    id_statistique = models.AutoField(primary_key=True)
    departement = models.CharField(max_length=10)
    type_statistique = models.CharField(max_length=16)
    valeur = models.IntegerField()
    date_enregistrement = models.DateField(auto_now_add=True)

    def __str__(self):
        return f"{self.type_statistique} ({self.departement})"
